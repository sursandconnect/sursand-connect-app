/* ==========================================================
   BASIC DATABASE FUNCTIONS
   ========================================================== */

function getDatabase() {

  return SpreadsheetApp
    .getActiveSpreadsheet();

}


function getSheet(sheetName) {

  const sheet =
    getDatabase()
      .getSheetByName(
        sheetName
      );


  if (!sheet) {

    throw new Error(
      "Sheet not found: " +
      sheetName
    );

  }


  return sheet;

}


/* ----------------------------------------------------------
   SAFE SHEET ACCESS

   Useful for optional sheets.
   ---------------------------------------------------------- */

function getSheetIfExists(sheetName) {

  return getDatabase()
    .getSheetByName(
      sheetName
    );

}



/* ==========================================================
   RESPONSE HELPERS
   ========================================================== */

function jsonResponse(data) {

  return ContentService
    .createTextOutput(

      JSON.stringify({

        success: true,

        data: data

      })

    )
    .setMimeType(
      ContentService.MimeType.JSON
    );

}


function errorResponse(message) {

  return ContentService
    .createTextOutput(

      JSON.stringify({

        success: false,

        error:
          String(
            message || "Unknown error."
          )

      })

    )
    .setMimeType(
      ContentService.MimeType.JSON
    );

}



/* ==========================================================
   SETTINGS
   ========================================================== */

function getSettings() {

  const sheet =
    getSheetIfExists(
      CONFIG.SHEETS.SETTINGS
    );


  if (!sheet) {

    return {};

  }


  const values =
    sheet
      .getDataRange()
      .getValues();


  if (values.length < 2) {

    return {};

  }


  const settings = {};


  /*
   * Supports:
   *
   * ID | KEY | VALUE
   *
   * and also:
   *
   * KEY | VALUE
   */

  for (
    let i = 1;
    i < values.length;
    i++
  ) {

    let key = "";
    let value = "";


    if (values[i].length >= 3) {

      key =
        String(
          values[i][1] || ""
        ).trim();

      value =
        values[i][2];

    }

    else {

      key =
        String(
          values[i][0] || ""
        ).trim();

      value =
        values[i][1];

    }


    if (key) {

      settings[key] =
        value;

    }

  }


  return settings;

}



/* ==========================================================
   SET / UPDATE SETTING
   ========================================================== */

function setSetting(
  key,
  value
) {

  key =
    String(
      key || ""
    ).trim();


  if (!key) {

    throw new Error(
      "Setting key is required."
    );

  }


  const sheet =
    getSheet(
      CONFIG.SHEETS.SETTINGS
    );


  const values =
    sheet
      .getDataRange()
      .getValues();


  /*
   * Existing setup normally stores:
   * ID | KEY | VALUE
   */

  const keyColumn =
    values[0] &&
    values[0].length >= 3
      ? 1
      : 0;


  const valueColumn =
    keyColumn === 1
      ? 2
      : 1;


  for (
    let i = 1;
    i < values.length;
    i++
  ) {

    if (
      String(
        values[i][keyColumn] || ""
      ).trim() === key
    ) {

      sheet
        .getRange(
          i + 1,
          valueColumn + 1
        )
        .setValue(
          value
        );


      return true;

    }

  }


  if (keyColumn === 1) {

    sheet.appendRow([
      Utilities.getUuid(),
      key,
      value
    ]);

  }

  else {

    sheet.appendRow([
      key,
      value
    ]);

  }


  return true;

}



/* ==========================================================
   SHEET HEADERS
   ========================================================== */

function getHeaders(sheetName) {

  const sheet =
    getSheet(
      sheetName
    );


  if (
    sheet.getLastColumn() < 1
  ) {

    return [];

  }


  return sheet
    .getRange(
      1,
      1,
      1,
      sheet.getLastColumn()
    )
    .getValues()[0]
    .map(function(header) {

      return String(
        header || ""
      ).trim();

    });

}



/* ==========================================================
   SHEET → JSON
   ========================================================== */

function sheetToJson(sheetName) {

  const sheet =
    getSheetIfExists(
      sheetName
    );


  if (!sheet) {

    return [];

  }


  const values =
    sheet
      .getDataRange()
      .getValues();


  if (values.length < 2) {

    return [];

  }


  const headers =
    values[0];


  return values
    .slice(1)
    .map(function(row) {

      const record = {};


      headers.forEach(
        function(header,index) {

          if (
            String(
              header || ""
            ).trim()
          ) {

            record[header] =
              normaliseOutputValue(
                row[index]
              );

          }

        }
      );


      return record;

    })
    .filter(
      function(record) {

        return Object
          .values(record)
          .some(
            function(value) {

              return (
                value !== "" &&
                value !== null
              );

            }
          );

      }
    );

}



/* ==========================================================
   NORMALISE DATE VALUES
   ========================================================== */

function normaliseOutputValue(value) {

  if (
    value instanceof Date
  ) {

    return Utilities
      .formatDate(
        value,
        CONFIG.TIMEZONE,
        "yyyy-MM-dd HH:mm:ss"
      );

  }


  return value;

}



/* ==========================================================
   PUBLIC RECORD FILTER
   ========================================================== */

function getActiveRecords(sheetName) {

  return sheetToJson(
    sheetName
  )
  .filter(
    function(record) {

      const active =
        String(
          record.Active || ""
        )
        .trim()
        .toLowerCase();


      const status =
        String(
          record.Status || ""
        )
        .trim()
        .toLowerCase();


      /* Explicitly inactive */

      if (
        active === "no" ||
        active === "false" ||
        active === "inactive" ||
        active === "disabled"
      ) {

        return false;

      }


      /*
       * IMPORTANT:
       * Pending submissions must never
       * become public automatically.
       */

      if (
        status === "pending" ||
        status === "rejected" ||
        status === "inactive" ||
        status === "disabled" ||
        status === "deleted" ||
        status === "new"
      ) {

        return false;

      }


      return true;

    }
  );

}



/* ==========================================================
   GENERATE ID
   ========================================================== */

function generateId(
  sheetName,
  prefix
) {

  const sheet =
    getSheet(
      sheetName
    );


  const values =
    sheet
      .getDataRange()
      .getValues();


  if (!prefix) {

    prefix =
      getIdPrefix(
        sheetName
      );

  }


  let highest = 0;


  if (values.length > 1) {

    const headers =
      values[0];


    const idColumn =
      headers.indexOf(
        "ID"
      );


    if (idColumn !== -1) {

      const regex =
        new RegExp(

          "^" +

          escapeRegExp(
            prefix
          ) +

          "(\\d+)$",

          "i"

        );


      for (
        let i = 1;
        i < values.length;
        i++
      ) {

        const id =
          String(
            values[i][idColumn] || ""
          )
          .trim();


        const match =
          id.match(
            regex
          );


        if (match) {

          const number =
            parseInt(
              match[1],
              10
            );


          if (
            number > highest
          ) {

            highest =
              number;

          }

        }

      }

    }

  }


  return (

    prefix +

    String(
      highest + 1
    )
    .padStart(
      4,
      "0"
    )

  );

}

/* ==========================================================
   ESCAPE REGEX
   ========================================================== */

function escapeRegExp(value) {

  return String(
    value || ""
  )
  .replace(
    /[.*+?^${}()|[\]\\]/g,
    "\\$&"
  );

}



/* ==========================================================
   GET FIRST AVAILABLE VALUE
   ========================================================== */

function firstValue(
  record,
  keys
) {

  for (
    let i = 0;
    i < keys.length;
    i++
  ) {

    const value =
      record[
        keys[i]
      ];


    if (
      value !== undefined &&
      value !== null &&
      String(value).trim() !== ""
    ) {

      return value;

    }

  }


  return "";

}



/* ==========================================================
   CLEAN PHONE
   ========================================================== */

function cleanPhoneNumber(phone) {

  return String(
    phone || ""
  )
  .replace(
    /\D/g,
    ""
  );

}



/* ==========================================================
   NORMALISE INDIAN WHATSAPP NUMBER
   ========================================================== */

function normaliseWhatsAppNumber(phone) {

  let number =
    cleanPhoneNumber(
      phone
    );


  if (
    number.length === 10
  ) {

    number =
      "91" +
      number;

  }


  return number;

}



/* ==========================================================
   GENERIC CONTACT ACTIONS
   ========================================================== */

function addContactActions(record) {

  const phone =
    cleanPhoneNumber(

      firstValue(
        record,
        [
          "Phone",
          "Mobile",
          "Contact Number",
          "Contact"
        ]
      )

    );


  const whatsapp =
    normaliseWhatsAppNumber(

      firstValue(
        record,
        [
          "WhatsApp",
          "WhatsApp Number",
          "Whatsapp Number",
          "Phone",
          "Mobile",
          "Contact Number"
        ]
      )

    );


  record.callUrl =
    phone
      ? "tel:" + phone
      : "";


  record.whatsappUrl =
    whatsapp
      ? "https://wa.me/" +
        whatsapp
      : "";


  return record;

}

/* ==========================================================
   SAFE DATE
   ========================================================== */

function parseDateSafe(value) {

  if (!value) {

    return 0;

  }


  const time =
    new Date(
      value
    )
    .getTime();


  return isNaN(time)
    ? 0
    : time;

}

/* ==========================================================
   LOCAL ADS SHEET
   ========================================================== */

function ensureLocalAdsSheet() {

  const db =
    getDatabase();

  let sheet =
    db.getSheetByName(
      CONFIG.SHEETS.LOCAL_ADS
    );

  if (!sheet) {

    sheet =
      db.insertSheet(
        CONFIG.SHEETS.LOCAL_ADS
      );

    sheet
      .getRange(
        1,1,1,10
      )
      .setValues([[
        "ID",
        "Title",
        "Description",
        "Image URL",
        "Link URL",
        "Active",
        "Start Date",
        "End Date",
        "Priority",
        "Created At"
      ]]);

    sheet.setFrozenRows(1);
  }

  return sheet;
}

/* ==========================================================
   GET API
   ========================================================== */

function doGet(e) {

  try {

    const parameters =
      e &&
      e.parameter
        ? e.parameter
        : {};


    const action =
      String(
        parameters.action ||
        "all"
      )
      .trim();


    switch(action) {


      /* ----------------------------------------------------
         MAIN
         ---------------------------------------------------- */

      case "all":

        return jsonResponse(
          getAllPublicData()
        );


      /* ----------------------------------------------------
         BUSINESSES
         ---------------------------------------------------- */

      case "businesses":

        return jsonResponse(
          getBusinessesWithActions()
        );


      /* ----------------------------------------------------
         LOCAL SERVICES
         ---------------------------------------------------- */

      case "localServices":

      case "services":

        return jsonResponse(
          prepareRecords(
            CONFIG.SHEETS.LOCAL_SERVICES
          )
        );


      /* ----------------------------------------------------
         EVENTS
         ---------------------------------------------------- */

      case "events":

        return jsonResponse(
          getEvents()
        );


      case "announcements":

        return jsonResponse(
          getAnnouncements()
        );


      /* ----------------------------------------------------
         EMERGENCY
         ---------------------------------------------------- */

      case "emergency":

        return jsonResponse(
          prepareRecords(
            CONFIG.SHEETS.EMERGENCY
          )
        );


      /* ----------------------------------------------------
         REPRESENTATIVES
         ---------------------------------------------------- */

      case "councillors":

      case "representatives":

        return jsonResponse(
          prepareRecords(
            CONFIG.SHEETS.COUNCILLORS
          )
        );


      /* ----------------------------------------------------
         IMPORTANT CONTACTS
         ---------------------------------------------------- */

      case "contacts":

        return jsonResponse(
          prepareRecords(
            CONFIG.SHEETS.CONTACTS
          )
        );


      /* ----------------------------------------------------
         EDUCATION
         ---------------------------------------------------- */

      case "schools":

      case "education":

        return jsonResponse(
          prepareRecords(
            CONFIG.SHEETS.SCHOOLS
          )
        );


      /* ----------------------------------------------------
         TRANSPORT
         ---------------------------------------------------- */

      case "transport":

        return jsonResponse(
          prepareRecords(
            CONFIG.SHEETS.TRANSPORT

/* ==========================================================
   APPEND RECORD USING SHEET HEADERS
   ========================================================== */

function appendRecord(
  sheetName,
  suppliedData,
  options
) {

  const sheet =
    getSheet(
      sheetName
    );


  const headers =
    getHeaders(
      sheetName
    );


  options =
    options || {};


  const data =
    suppliedData || {};


  let generatedId =
    "";


  const row =
    headers.map(
      function(header) {

        const normal =
          normaliseHeader(
            header
          );


        /* ID */

        if (
          normal === "id"
        ) {

          generatedId =
            generateId(
              sheetName
            );


          return generatedId;

        }


        /* Dates */

        if (
          normal === "timestamp" ||
          normal === "createdat" ||
          normal === "submittedat"
        ) {

          return new Date();

        }


        /* Status */

        if (
          normal === "status" &&
          options.defaultStatus !== undefined
        ) {

          return (
            lookupInputValue(
              data,
              header
            ) ||
            options.defaultStatus
          );

        }


        /* Active */

        if (
          normal === "active" &&
          options.defaultActive !== undefined
        ) {

          return (
            lookupInputValue(
              data,
              header
            ) ||
            options.defaultActive
          );

        }


        return lookupInputValue(
          data,
          header
        );

      }
    );


  sheet.appendRow(
    row
  );


  return {

    id:
      generatedId,

    row:
      row

  };

}



/* ==========================================================
   NORMALISE HEADER
   ========================================================== */

function normaliseHeader(value) {

  return String(
    value || ""
  )
  .toLowerCase()
  .replace(
    /[^a-z0-9]/g,
    ""
  );

}

function ensureRequestSheets() {

  const db = getDatabase();

  const specs = [
    {
      name: CONFIG.SHEETS.SERVICE_REGISTRATIONS,
      headers: ["ID","Service Person Name","Main Category","Service Type","Phone","WhatsApp","Ward","Address","Description","Experience","Availability","Image URL","Map URL","Status","Created At"]
    },
    {
      name: CONFIG.SHEETS.CHANGE_MAKER_REQUESTS,
      headers: ["ID","Full Name","Mobile","Email","Ward","Address","Area of Interest","Occupation","Your Idea / Message","Status","Created At"]
    },
    {
      name: CONFIG.SHEETS.ADMIN_NOTIFICATIONS,
      headers: ["ID","Request Type","Request ID","Title","Message","Status","Created At"]
    }
  ];

  specs.forEach(function(spec) {
    let sheet = db.getSheetByName(spec.name);
    if (!sheet) {
      sheet = db.insertSheet(spec.name);
      sheet.getRange(1,1,1,spec.headers.length).setValues([spec.headers]);
      sheet.setFrozenRows(1);
    }
  });

  return true;
}

      "Request Type": type,

      "Title": title,
      "Message": message,
      "Status": "New"
    },
    {
      defaultStatus: "New"
    }
  );
}

function clearAdminRequestNotification_(requestId) {

  const sheet = getSheetIfExists(CONFIG.SHEETS.ADMIN_NOTIFICATIONS);
  if (!sheet) return true;

  const values = sheet.getDataRange().getValues();
  if (values.length < 2) return true;

  const headers = values[0];
  const col = headers.indexOf("Request ID");
  if (col === -1) return true;

  for (let i = values.length - 1; i >= 1; i--) {
    if (String(values[i][col]) === String(requestId)) {
      sheet.deleteRow(i + 1);
    }
  }

  return true;
}

/* ==========================================================
   IMAGE / FILE UPLOAD
   ========================================================== */

function processIncomingImage(
  data,
  prefix
) {

  const base64 =
    data.imageBase64 ||
    data.fileBase64 ||
    "";


  if (!base64) {

    return (
      data.image ||
      data.imageUrl ||
      ""
    );

  }


  const fileName =
    data.imageName ||
    data.fileName ||
    (
      prefix +
      "-" +
      new Date().getTime() +
      ".jpg"
    );


  const mimeType =
    data.imageMimeType ||
    data.fileMimeType ||
    "image/jpeg";


  return saveBase64File(
    base64,
    fileName,
    mimeType,
    prefix
  );

}



/* ==========================================================
   SAVE BASE64 FILE TO DRIVE
   ========================================================== */

function saveBase64File(
  base64,
  fileName,
  mimeType,
  folderPrefix
) {

  let cleanBase64 =
    String(
      base64 || ""
    );


  if (
    cleanBase64.indexOf(
      "base64,"
    ) !== -1
  ) {

    cleanBase64 =
      cleanBase64
        .split(
          "base64,"
        )[1];

  }


  const bytes =
    Utilities
      .base64Decode(
        cleanBase64
      );


  const blob =
    Utilities
      .newBlob(
        bytes,
        mimeType,
        fileName
      );


  const folder =
    getUploadFolder(
      folderPrefix
    );


  const file =
    folder
      .createFile(
        blob
      );


  /*
   * Public image access is useful for
   * public business/place cards.
   *
   * If Workspace restrictions prevent
   * public sharing, the file is still saved.
   */

  try {

    file.setSharing(

      DriveApp.Access.ANYONE_WITH_LINK,

      DriveApp.Permission.VIEW

    );

  }

  catch(error) {

    Logger.log(
      "Public sharing unavailable: " +
      error.message
    );

  }


  return createDriveViewUrl(
    file.getId()
  );

}



/* ==========================================================
   UPLOAD FOLDER
   ========================================================== */

function getUploadFolder(prefix) {

  const rootName =
    "Sursand Connect Uploads";


  let root;


  const folders =
    DriveApp
      .getFoldersByName(
        rootName
      );


  if (
    folders.hasNext()
  ) {

    root =
      folders.next();

  }

  else {

    root =
      DriveApp
        .createFolder(
          rootName
        );

  }


  const childName =
    String(
      prefix || "GENERAL"
    )
    .toUpperCase();


  const children =
    root
      .getFoldersByName(
        childName
      );


  if (
    children.hasNext()
  ) {

    return children.next();

  }


  return root
    .createFolder(
      childName
    );

}



/* ==========================================================
   DRIVE IMAGE URL
   ========================================================== */

function createDriveViewUrl(fileId) {

  return (
    "https://drive.google.com/uc?export=view&id=" +
    encodeURIComponent(
      fileId
    )
  );

}

      type:
        "Agriculture"
    },

    {
      sheet:
        CONFIG.SHEETS.USEFUL_LINKS,
      type:
        "Useful Website"
    }

  ];


  const results = [];


  sources.forEach(
    function(source) {

      const records =
        getActiveRecords(
          source.sheet
        );


      records.forEach(
        function(record) {

          const text =
            Object
              .values(record)
              .join(" ")
              .toLowerCase();


          if (
            text.indexOf(
              query
            ) !== -1
          ) {

            results.push({

              type:
                source.type,

              id:
                record.ID ||
                "",

              title:
                getRecordTitle(
                  record
                ),

              data:
                record

            });

          }

        }
      );

    }
  );


  return results;

}

/* ==========================================================
   ADMIN PASSWORD HASH
   ========================================================== */

function hashAdminPassword(password) {

  const digest =
    Utilities.computeDigest(

      Utilities.DigestAlgorithm.SHA_256,

      String(
        password || ""
      ),

      Utilities.Charset.UTF_8

    );


  return digest
    .map(
      function(byte) {

        const value =
          byte < 0
            ? byte + 256
            : byte;


        return (
          "0" +
          value.toString(16)
        ).slice(-2);

      }
    )
    .join("");

}



/* ==========================================================
   VERIFY ADMIN LOGIN
   ========================================================== */

function verifyAdminLogin(
  user,
  password
) {

  const properties =
    PropertiesService
      .getScriptProperties();


  const savedUser =
    properties.getProperty(
      "SC_ADMIN_USER"
    );


  const savedPasswordHash =
    properties.getProperty(
      "SC_ADMIN_PASSWORD_HASH"
    );


  if (
    !savedUser ||
    !savedPasswordHash
  ) {

    throw new Error(
      "Admin login is not configured in Script Properties."
    );

  }


  const enteredHash =
    hashAdminPassword(
      password
    );


  return (

    String(user || "") ===
      String(savedUser)

    &&

    enteredHash ===
      savedPasswordHash

  );

}



/* ==========================================================
   CREATE ADMIN SESSION
   ========================================================== */

function createAdminSession() {

  const token =

    Utilities.getUuid() +

    "-" +

    Utilities.getUuid();


  /*
   * Six hours
   */

  CacheService
    .getScriptCache()
    .put(

      "ADMIN_SESSION_" +
      token,

      "1",

      21600

    );


  return token;

}



/* ==========================================================
   VERIFY ADMIN SESSION
   ========================================================== */

function verifyAdminSession(token) {

  token =
    String(
      token || ""
    )
    .trim();


  if (!token) {

    throw new Error(
      "Admin session required."
    );

  }


  const session =
    CacheService
      .getScriptCache()
      .get(

        "ADMIN_SESSION_" +
        token

      );


  if (!session) {

    throw new Error(
      "Admin session expired. Please login again."
    );

  }


  return true;

}

/* ==========================================================
   ADMIN DATA
   ========================================================== */

function adminDataApi(data) {

  verifyAdminSession(
    data.token
  );


  const section =
    String(
      data.section || ""
    )
    .trim();


  ensureRequestSheets();

  const config =
    ADMIN_SECTION_MAP[
      section
    ];

  if (
    section === "localAds"
  ) {
    ensureLocalAdsSheet();
  }




  if (!config) {

    throw new Error(
      "Unknown Admin Panel section: " +
      section
    );

  }


  /* --------------------------------------------------------
     BUSINESSES:
     registrations + published listings
     -------------------------------------------------------- */

  if (
    config.special ===
    "businesses"
  ) {

    const registrations =
      sheetToJson(
        CONFIG.SHEETS.BUSINESS_REGISTRATIONS
      )
      .map(
        function(record) {

          record._source =
            "registration";

          return record;

        }
      );


    const published =
      sheetToJson(
        CONFIG.SHEETS.BUSINESSES
      )
      .map(
        function(record) {

          record._source =
            "published";

          return record;

        }
      );


    return jsonResponse(
      registrations.concat(
        published
      )
    );

  }



  if (config.special === "services") {

    ensureRequestSheets();

    const requests =
      sheetToJson(CONFIG.SHEETS.SERVICE_REGISTRATIONS)
      .map(function(record){ record._source = "registration"; return record; });

    const published =
      sheetToJson(CONFIG.SHEETS.LOCAL_SERVICES)
      .map(function(record){ record._source = "published"; return record; });

    return jsonResponse(requests.concat(published));
  }

  if (config.special === "changeMakers") {

    ensureRequestSheets();

    const requests =
      sheetToJson(CONFIG.SHEETS.CHANGE_MAKER_REQUESTS)
      .map(function(record){ record._source = "registration"; return record; });

    const published =
      sheetToJson(CONFIG.SHEETS.CHANGE_MAKERS)
      .map(function(record){ record._source = "published"; return record; });

    return jsonResponse(requests.concat(published));
  }

  /* --------------------------------------------------------
     EVENTS + ANNOUNCEMENTS
     -------------------------------------------------------- */

  if (
    config.special ===
    "events"
  ) {

    const events =
      sheetToJson(
        CONFIG.SHEETS.EVENTS
      )
      .map(
        function(record) {

          record._source =
            "event";

          return record;

        }
      );


    const announcements =
      sheetToJson(
        CONFIG.SHEETS.ANNOUNCEMENTS
      )
      .map(
        function(record) {

          record._source =
            "announcement";

          return record;

        }
      );


    return jsonResponse(
      events.concat(
        announcements
      )
    );

  }


  /* --------------------------------------------------------
     CITY CONNECT
     -------------------------------------------------------- */

  if (
    config.special ===
    "cityConnect"
  ) {

    return jsonResponse(
      getCityConnectGroups()
    );

  }


  return jsonResponse(
    sheetToJson(
      config.sheet
    )
  );

}



/* ==========================================================
   ADMIN SAVE
   Used by new Admin Panel for Add/Edit
   ========================================================== */

function adminSaveApi(data) {

  verifyAdminSession(
    data.token
  );


  const section =
    String(
      data.section || ""
    )
    .trim();


  const operation =
    String(
      data.operation ||
      "add"
    )
    .trim()
    .toLowerCase();


  const record =
    data.data ||
    data.record ||
    {};


  const id =
    String(
      data.id || ""
    );



  /* --------------------------------------------------------
     APP SETTINGS
     -------------------------------------------------------- */

  if (
    section === "settings"
  ) {

    const key =
      String(
        record.KEY ||
        record.Key ||
        record.key ||
        ""
      ).trim();

    const value =
      record.VALUE !== undefined
        ? record.VALUE
        : (
            record.Value !== undefined
              ? record.Value
              : record.value
          );

    if (!key) {
      throw new Error(
        "Setting KEY is required."
      );
    }

    setSetting(
      key,
      value
    );

    return jsonResponse({
      message:
        "Setting updated successfully."
    });
  }


  /* --------------------------------------------------------
     CITY CONNECT
     -------------------------------------------------------- */

  if (
    section ===
    "cityConnect"
  ) {

    return adminSaveCityConnect(
      operation,
      id,
      record
    );

  }


  const sheetName =
    resolveAdminSheetForSave(
      section,
      id,
      record
    );


  if (!sheetName) {

    throw new Error(
      "Admin section is not configured for saving."
    );

  }


  if (
    operation === "add"
  ) {

    const result =
      appendRecord(

        sheetName,

        record,

        {
          defaultStatus:
            defaultAdminStatus(
              section
            ),

          defaultActive:
            "Yes"
        }

      );


    return jsonResponse({

      message:
        "Record added successfully.",

      id:
        result.id

    });

  }


  if (
    operation === "edit"
  ) {

    updateRecordById(
      sheetName,
      id,
      record
    );


    return jsonResponse({

      message:
        "Record updated successfully.",

      id:
        id

    });

  }


  throw new Error(
    "Unknown save operation: " +
    operation
  );

}



/* ==========================================================
   DEFAULT ADMIN STATUS

function defaultAdminStatus(section) {

  if (
    section === "changeMakers"
  ) {

    return "Approved";

  }


  return "Active";

}



/* ==========================================================
   RESOLVE SHEET FOR ADMIN SAVE
   ========================================================== */

function resolveAdminSheetForSave(
  section,
  id,
  record
) {

  /* Businesses */

  if (
    section === "businesses"
  ) {

    if (
      String(id)
        .toUpperCase()
        .indexOf("BR") === 0
    ) {

      return CONFIG
        .SHEETS
        .BUSINESS_REGISTRATIONS;

    }


    return CONFIG
      .SHEETS
      .BUSINESSES;

  }



  if (section === "services") {
    return String(id).toUpperCase().indexOf("SR") === 0
      ? CONFIG.SHEETS.SERVICE_REGISTRATIONS
      : CONFIG.SHEETS.LOCAL_SERVICES;
  }

  if (section === "changeMakers") {
    return String(id).toUpperCase().indexOf("CMR") === 0
      ? CONFIG.SHEETS.CHANGE_MAKER_REQUESTS
      : CONFIG.SHEETS.CHANGE_MAKERS;
  }

  /* Events / Announcements */

  if (
    section === "events"
  ) {

    if (
      String(id)
        .toUpperCase()
        .indexOf("A") === 0
    ) {

      return CONFIG
        .SHEETS
        .ANNOUNCEMENTS;

    }


    const category =
      String(

        record.Type ||
        record.Category ||
        ""

      )
      .toLowerCase();


    if (
      category ===
      "announcement"
    ) {

      return CONFIG
        .SHEETS
        .ANNOUNCEMENTS;

    }


    return CONFIG
      .SHEETS
      .EVENTS;

  }


  const config =
    ADMIN_SECTION_MAP[
      section
    ];


  return (
    config &&
    config.sheet
      ? config.sheet
      : ""
  );

}



/* ==========================================================
   ADMIN UPDATE
   Approve / Reject / Delete
   ========================================================== */

function adminUpdateApi(data) {

  verifyAdminSession(
    data.token
  );


  const section =
    String(
      data.section || ""
    )
    .trim();


  const operation =
    String(
      data.operation || ""
    )
    .trim()
    .toLowerCase();


  const id =
    String(
      data.id || ""
    )
    .trim();


  if (!id) {

    throw new Error(
      "Record ID is required."
    );

  }


  /* --------------------------------------------------------
     BUSINESS APPROVAL

     BRxxxx → publish to Businesses
     -------------------------------------------------------- */

  if (
    section === "businesses" &&
    operation === "approve" &&
    id.toUpperCase().indexOf("BR") === 0
  ) {

    return jsonResponse(
      approveBusinessRegistration(
        id
      )
    );

  }



  if (
    section === "services" &&
    operation === "approve" &&
    id.toUpperCase().indexOf("SR") === 0
  ) {
    return jsonResponse(approveServiceRegistration_(id));
  }

  if (
    section === "changeMakers" &&
    operation === "approve" &&
    id.toUpperCase().indexOf("CMR") === 0
  ) {
    return jsonResponse(approveChangeMakerRequest_(id));
  }

  if (
    operation === "reject" &&
    (
      id.toUpperCase().indexOf("BR") === 0 ||
      id.toUpperCase().indexOf("SR") === 0 ||
      id.toUpperCase().indexOf("CMR") === 0
    )
  ) {
    const requestSheet =
      id.toUpperCase().indexOf("BR") === 0
        ? CONFIG.SHEETS.BUSINESS_REGISTRATIONS
        : (
            id.toUpperCase().indexOf("SR") === 0
              ? CONFIG.SHEETS.SERVICE_REGISTRATIONS
              : CONFIG.SHEETS.CHANGE_MAKER_REQUESTS
          );

    deleteRecordById(requestSheet,id);
    clearAdminRequestNotification_(id);

    return jsonResponse({
      message: "Request rejected and removed."
    });
  }

  const sheetName =
    resolveAdminSheetForSave(
      section,
      id,
      data.data || {}
    );


  if (!sheetName) {

    throw new Error(
      "Unable to determine sheet."
    );

  }


  switch(operation) {


    case "approve":

      changeRecordStatus(
        sheetName,
        id,
        "Approved"
      );


      return jsonResponse({

        message:
          "Record approved successfully."

      });


    case "reject":

      changeRecordStatus(
        sheetName,
        id,
        "Rejected"
      );


      return jsonResponse({

        message:
          "Record rejected successfully."

      });


    case "activate":

      changeRecordStatus(
        sheetName,
        id,
        "Active"
      );


      return jsonResponse({

        message:
          "Record activated successfully."

      });


    case "deactivate":

      changeRecordStatus(
        sheetName,
        id,
        "Inactive"
      );


      return jsonResponse({

        message:
          "Record deactivated successfully."

      });


    case "delete":

      deleteRecordById(
        sheetName,
        id
      );


      return jsonResponse({

        message:
          "Record deleted successfully."

      });


    default:

      throw new Error(
        "Unknown admin operation: " +
        operation
      );

  }

}




/* ==========================================================
   ADMIN RECORD ID COLUMN
   ========================================================== */

function findRecordIdColumn_(headers) {

  const candidates = [
    "ID",
    "User ID",
    "Id",
    "id"
  ];

  for (
    let i = 0;
    i < candidates.length;
    i++
  ) {

    const index =
      headers.indexOf(
        candidates[i]
      );

    if (index !== -1) {
      return index;
    }
  }

  return -1;
}

/* ==========================================================
   UPDATE RECORD BY ID
   ========================================================== */

function updateRecordById(
  sheetName,
  recordId,
  data
) {

  const sheet =
    getSheet(
      sheetName
    );


  const values =
    sheet
      .getDataRange()
      .getValues();


  if (
    values.length < 2
  ) {

    throw new Error(
      "No records found."
    );

  }


  const headers =
    values[0];


  const idColumn =
    findRecordIdColumn_(
      headers
    );


  if (
    idColumn === -1
  ) {

    throw new Error(
      "ID column not found in " +
      sheetName
    );

  }


  for (
    let row = 1;
    row < values.length;
    row++
  ) {

    if (
      String(
        values[row][idColumn]
      ) ===
      String(
        recordId
      )
    ) {

      headers.forEach(
        function(header,column) {

          if (
            normaliseHeader(header) === "id" || normaliseHeader(header) === "userid"
          ) {

            return;

          }


          const value =
            lookupInputValue(
              data,
              header
            );


          /*
           * Only update if Admin Panel
           * actually supplied this field.
           */

          const hasDirect =
            data[header] !== undefined;


          const hasNormalised =
            Object
              .keys(data)
              .some(
                function(key) {

                  return (
                    normaliseHeader(key) ===
                    normaliseHeader(header)
                  );

                }
              );


          if (
            hasDirect ||
            hasNormalised
          ) {

            sheet
              .getRange(
                row + 1,
                column + 1
              )
              .setValue(
                value
              );

          }

        }
      );


      const updatedColumn =
        headers.indexOf(
          "Updated At"
        );


      if (
        updatedColumn !== -1
      ) {

        sheet
          .getRange(
            row + 1,
            updatedColumn + 1
          )
          .setValue(
            new Date()
          );

      }


      return true;

    }

  }


  throw new Error(
    "Record not found: " +
    recordId
  );

}



/* ==========================================================
   CHANGE STATUS
   ========================================================== */

function changeRecordStatus(
  sheetName,
  id,
  status
) {

  const sheet =
    getSheet(
      sheetName
    );


  const values =
    sheet
      .getDataRange()
      .getValues();


  const headers =
    values[0] || [];


  const idColumn =
    findRecordIdColumn_(
      headers
    );


  let statusColumn =
    headers.indexOf(
      "Status"
    );


  if (
    idColumn === -1
  ) {

    throw new Error(
      "ID column not found."
    );

  }


  /*
   * If Status column doesn't exist,
   * create it automatically.
   */

  if (
    statusColumn === -1
  ) {

    statusColumn =
      headers.length;


    sheet
      .getRange(
        1,
        statusColumn + 1
      )
      .setValue(
        "Status"
      );

  }


  for (
    let i = 1;
    i < values.length;
    i++
  ) {

    if (
      String(
        values[i][idColumn]
      ) ===
      String(id)
    ) {

      sheet
        .getRange(
          i + 1,
          statusColumn + 1
        )
        .setValue(
          status
        );


      return true;

    }

  }


  throw new Error(
    "Record not found."
  );

}



/* ==========================================================

) {

  const sheet =
    getSheet(
      sheetName
    );


  const values =
    sheet
      .getDataRange()
      .getValues();


  const headers =
    values[0] || [];


  const idColumn =
    findRecordIdColumn_(
      headers
    );


  if (
    idColumn === -1
  ) {

    throw new Error(

    );

  }


  for (
    let i = 1;
    i < values.length;
    i++
  ) {

    if (

        values[i][idColumn]

      );

    }

  }


  throw new Error(
    "Record not found."
  );

}




/* ==========================================================
   SERVICE REQUEST APPROVAL
   ========================================================== */

function approveServiceRegistration_(requestId) {

  const rows = sheetToJson(CONFIG.SHEETS.SERVICE_REGISTRATIONS);
  const record = rows.find(function(r){ return String(r.ID) === String(requestId); });
  if (!record) throw new Error("Service registration not found.");

  const added = appendRecord(
    CONFIG.SHEETS.LOCAL_SERVICES,
    record,
    { defaultStatus: "Approved", defaultActive: "Yes" }
  );

  deleteRecordById(CONFIG.SHEETS.SERVICE_REGISTRATIONS, requestId);
  clearAdminRequestNotification_(requestId);

  return {
    message: "Service approved and published successfully.",
    serviceId: added.id,
    requestId: requestId

function approveBusinessRegistration(
  registrationId
) {

  const sourceSheet =
    getSheet(
      CONFIG.SHEETS.BUSINESS_REGISTRATIONS
    );


  const targetSheet =
    getSheet(
      CONFIG.SHEETS.BUSINESSES
    );


  const values =
    sourceSheet
      .getDataRange()
      .getValues();


  if (
    values.length < 2
  ) {

    throw new Error(
      "No business registrations found."
    );

  }


  const headers =
    values[0];


  const idColumn =
    headers.indexOf(
      "ID"
    );


  if (
    idColumn === -1
  ) {

    throw new Error(
      "ID column not found in Business Registrations."
    );

  }


  for (
    let row = 1;
    row < values.length;
    row++
  ) {

    if (
      String(
        values[row][idColumn]
      ) !==
      String(
        registrationId
      )
    ) {

      continue;

    }


    const registration = {};


    headers.forEach(
      function(header,index) {

        registration[header] =
          values[row][index];

      }
    );


    const name =
      firstValue(
        registration,
        [
          "Shop / Business Name",
          "Business Name"
        ]
      );


    const phone =
      firstValue(
        registration,
        [
          "Phone",
          "Mobile"
        ]
      );


    /* Duplicate protection */

    const duplicate =
      sheetToJson(
        CONFIG.SHEETS.BUSINESSES
      )
      .some(
        function(item) {

          const existingName =
            firstValue(
              item,
              [
                "Shop / Business Name",
                "Business Name"
              ]
            );


          return (

            String(existingName)
              .trim()
              .toLowerCase()
              ===
            String(name)
              .trim()
              .toLowerCase()

            &&

            cleanPhoneNumber(
              item.Phone
            )
            ===
            cleanPhoneNumber(
              phone
            )

          );

        }
      );


    if (duplicate) {

      throw new Error(
        "This shop/business is already published."
      );

    }


    /*
     * Append using target headers.
     * Existing field names are mapped automatically.
     */

    const businessData = {

      businessName:
        name,

      owner:
        firstValue(
          registration,
          [
            "Owner / Contact Person",
            "Owner Name",
            "Owner"
          ]
        ),

      category:
        registration.Category ||
        "",

      phone:
        phone,

      whatsapp:
        registration.WhatsApp ||
        phone,

      ward:
        registration.Ward ||
        "",

      address:
        registration.Address ||
        "",

      description:
        registration.Description ||
        "",

      catalogue:
        firstValue(
          registration,
          [
            "WhatsApp Catalogue Link",
            "Catalogue URL"
          ]
        ),

      website:
        registration.Website ||
        "",

      map:
        firstValue(
          registration,
          [
            "Google Maps Link",
            "Google Map Link",
            "Map URL"
          ]
        ),

      latitude:
        registration.Latitude ||
        "",

      longitude:
        registration.Longitude ||
        "",

      image:
        firstValue(
          registration,
          [
            "Business Image",
            "Business Image URL",
            "Image URL",
            "Image"
          ]
        ),

      status:
        "Approved",

      active:
        "Yes"

    };


    const added =
      appendRecord(

        CONFIG.SHEETS.BUSINESSES,

        businessData,

        {
          defaultStatus:
            "Approved",

          defaultActive:
            "Yes"
        }

      );


    deleteRecordById(
      CONFIG.SHEETS.BUSINESS_REGISTRATIONS,
      registrationId
    );

    clearAdminRequestNotification_(
      registrationId
    );


    return {

      message:
        "Shop / Business approved and published successfully.",

      businessId:
        added.id,

      registrationId:
        registrationId

    };

  }


  throw new Error(
    "Business registration not found."
  );

}

/* ==========================================================
   ADMIN BULK UPDATE
   ========================================================== */

function adminBulkUpdateApi(data) {

  verifyAdminSession(
    data.token
  );

  const section =
    String(
      data.section || ""
    ).trim();

  const operation =
    String(
      data.operation || ""
    ).trim().toLowerCase();

  const ids =
    Array.isArray(
      data.ids
    )
      ? data.ids
      : [];

  if (
    operation !== "approve" &&
    operation !== "reject"
  ) {
    throw new Error(
      "Bulk operation must be approve or reject."
    );
  }

  if (!ids.length) {
    throw new Error(
      "No records selected."
    );
  }

  let completed = 0;
  let failed = 0;

  ids.forEach(function(id) {

    try {

      const result =
        adminUpdateApi({
          token:
            data.token,
          section:
            section,
          operation:
            operation,
          id:
            id
        });

      if (result) {
        completed++;
      }

    }
    catch(error) {
      failed++;
    }
  });

  return jsonResponse({
    message:
      completed +
      " record(s) " +
      operation +
      "d successfully" +
      (
        failed
          ? "; " +
            failed +
            " failed."
          : "."
      ),
    completed:
      completed,
    failed:
      failed
  });
}

const SC_STAFF_ADMIN_SHEET = "Staff Admins";
const SC_STAFF_SESSION_SHEET = "Staff Admin Sessions";
const SC_STAFF_SESSION_HOURS = 8;

function setupStaffAdminSystem() {
  const db = getDatabase();

  let admins = db.getSheetByName(SC_STAFF_ADMIN_SHEET);
  if (!admins) {
    admins = db.insertSheet(SC_STAFF_ADMIN_SHEET);
    admins.getRange(1,1,1,13).setValues([[
      "Admin ID","Name","Mobile","Email","Ward","Address",
      "Password Salt","Password Hash","Status","Created At",
      "Updated At","Last Login","Created By"
    ]]);
    admins.setFrozenRows(1);
  }

  let sessions = db.getSheetByName(SC_STAFF_SESSION_SHEET);
  if (!sessions) {
    sessions = db.insertSheet(SC_STAFF_SESSION_SHEET);
    sessions.getRange(1,1,1,6).setValues([[
      "Token","Admin ID","Created At","Expires At","Active","Role"
    ]]);
    sessions.setFrozenRows(1);
  }

  ensureRequestSheets();
  return true;
}

function staffRows_(sheetName) {
  const sh = getDatabase().getSheetByName(sheetName);
  if (!sh) throw new Error("Run setupStaffAdminSystem() first.");
  const v = sh.getDataRange().getValues();
  if (v.length < 2) return [];
  const h = v[0];
  return v.slice(1).map(function(row,i){
    const o={__row:i+2};
    h.forEach(function(k,c){o[k]=row[c]});
    return o;
  });
}

function staffCleanPhone_(v) {
  let d=String(v||"").replace(/\D/g,"");
  if (d.length===12 && d.indexOf("91")===0) d=d.substring(2);
  return d;
}
function staffCleanEmail_(v){return String(v||"").trim().toLowerCase()}
function staffHash_(password,salt){
  let value=String(salt||"")+"|"+String(password||"");
  for(let i=0;i<1500;i++){
    const bytes=Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256,value,Utilities.Charset.UTF_8);
    value=bytes.map(function(b){const n=b<0?b+256:b;return (n<16?"0":"")+n.toString(16)}).join("");
  }
  return value;
}
function staffToken_(){return Utilities.getUuid().replace(/-/g,"")+Utilities.getUuid().replace(/-/g,"")}

function staffPublic_(a){
  return {
    id:String(a["Admin ID"]||""),
    name:String(a.Name||""),
    mobile:String(a.Mobile||""),
    email:String(a.Email||""),
    ward:String(a.Ward||""),
    address:String(a.Address||""),
    status:String(a.Status||"Active"),
    role:"staff"
  };
}

function staffFind_(login){
  const raw=String(login||"").trim(), phone=staffCleanPhone_(raw), email=staffCleanEmail_(raw);
  return staffRows_(SC_STAFF_ADMIN_SHEET).find(function(a){
    return (phone && staffCleanPhone_(a.Mobile)===phone) ||
           (email && staffCleanEmail_(a.Email)===email) ||
           String(a["Admin ID"]||"").toLowerCase()===raw.toLowerCase();
  });
}

function staffSession_(adminId){
  const token=staffToken_(), now=new Date(), exp=new Date(now.getTime()+SC_STAFF_SESSION_HOURS*3600000);
  getDatabase().getSheetByName(SC_STAFF_SESSION_SHEET).appendRow([token,adminId,now,exp,"Yes","staff"]);
  return {token:token,expiresIn:SC_STAFF_SESSION_HOURS*3600};
}

function verifyStaffSession_(token){
  setupStaffAdminSystem();
  const s=staffRows_(SC_STAFF_SESSION_SHEET).find(function(x){
    return String(x.Token)===String(token||"") && String(x.Active).toLowerCase()!=="no";
  });
  if(!s) throw new Error("Invalid admin session.");
  const exp=new Date(s["Expires At"]);
  if(isNaN(exp.getTime())||exp.getTime()<Date.now()) throw new Error("Admin session expired.");
  const a=staffRows_(SC_STAFF_ADMIN_SHEET).find(function(x){return String(x["Admin ID"])===String(s["Admin ID"])});
  if(!a||String(a.Status||"").toLowerCase()!=="active") throw new Error("Admin account is disabled.");
  return a;
}

function staffAdminLoginApi(data){
  setupStaffAdminSystem();
  const a=staffFind_(data.login||data.user);
  if(!a) throw new Error("Invalid admin login.");
  if(String(a.Status||"").toLowerCase()!=="active") throw new Error("Admin account is disabled.");
  const hash=staffHash_(data.password,String(a["Password Salt"]||""));

  const session=staffSession_(a["Admin ID"]);
  return jsonResponse({token:session.token,expiresIn:session.expiresIn,admin:staffPublic_(a)});
}

function staffAdminSaveApi(data){
  verifyStaffSession_(data.token);
  const section=String(data.section||""), op=String(data.operation||"add"), rec=data.data||{}, id=String(data.id||"");

  if(!["events","healthcare","education"].includes(section)) throw new Error("This admin cannot modify that module.");

  let sheetName="";
  if(section==="events"){
    sheetName=String(rec["Record Type"]||"Event")==="Announcement"?CONFIG.SHEETS.ANNOUNCEMENTS:CONFIG.SHEETS.EVENTS;

  return jsonResponse(updateRecordById(sheetName,id,rec));
